Let $F_n$ be the $n^\text{th}$ term in the Fibonacci sequence, which is defined as follows:

\[
  F_n = \begin{cases}
    1                 & \text{if } n = 1 \\
    1                 & \text{if } n = 2 \\
    F_{n-1} + F_{n-2} & \text{if } n \ge 3
  \end{cases}
\]

Then, let \{$O_n$\} be the sequence of the Fibonacci sequence's \textit{odd}-indexed terms and
\{$E_n$\} be the sequence of its \textit{even}-indexed terms.

Let's observe the sums of these two sequences.

\begin{proposition}
  \[
    \forall m \in \mathbb{N}, \sum_{j=1}^{m} F_{2j-1} = F_{2m}
  \]
\end{proposition}

\begin{proof} The proof is by induction on $m$.

  \emptyline

  \textbf{BASE CASE:} Consider the case for $m=1$. Observe,

  \[\begin{aligned}
    \sum_{j=1}^1 F_{2j-1} &= F_{2(1)-1} \\
                          &= F_{2-1} \\
                          &= F_1 \\
                          &= 1
  \end{aligned}\]

  and,

  \[
    F_{2(1)} = F_2 = 1
  \]

  Thus, the result holds for the base case $m=1$.

  \emptyline

  \textbf{INDUCTION:} Suppose the result holds for some natural number $k$. That is,

  \[
    \sum_{j=1}^{k} F_{2j-1} = F_{2k}
  \]

  Now consider the case for $k+1$. Observe,

  \begin{align}
    \sum_{j=1}^{k+1} F_{2j-1} &= \sum_{j=1}^k (F_{2j-1}) + F_{2(k+1)-1} \notag \\
                    &= F_{2k} + F_{2(k+1)-1} \tag{by the inductive hypothesis} \\
                    &= F_{2k} + F_{2k+1} \notag \\
                    &= F_{2k+2} \tag{by the definition of $F_n$} \\
                    &= F_{2(k+1)} \notag
  \end{align}

  Notice that the substitution of $F_{2k+2}$ for $F_{2k} + F_{2k+1}$ is valid because the index $2k+2$ is always greater than 3. So $F_{2k+2}$ sits within the $n \ge 3$ piece of $F_n$, where $F_n = F_{n-1} + F_{n-2}$.

  The result, $F_{2(k+1)}$, has the form $F_{2m}$ for $m = k+1$. Thus, the result holds in the inductive case. Therefore, by the principle of mathematical induction, the result holds for any natural number $m$.

\end{proof}

\begin{proposition}
  \[
    \forall m \in \mathbb{N}, \sum_{j=1}^{m} F_{2j} = F_{2m+1} - 1
  \]
\end{proposition}

\begin{proof} The proof is by induction on $m$.

  \emptyline

  \textbf{BASE CASE:} Consider the case where $m=1$. Observe,

  \[\begin{aligned}
    \sum_{j=1}^1 F_{2j} &= F_{2(1)} \\
                        &= F_2 \\
                        &= 1
  \end{aligned}\]

  and,

  \[\begin{aligned}
    F_{2(1)+1}-1 &= F_3 - 1 \\
                 &= 2 - 1 \\
                 &= 1
  \end{aligned}\]

  Therefore, the result holds for the base case $m=1$.

  \emptyline

  \textbf{INDUCTION:} Suppose the result holds for some natural number $k$. That is,

  \[
    \sum_{2j}^{k} = F_{2k+1}-1
  \]

  Consider the case for $k+1$. Observe,

  \begin{align}
    \sum_{j=1}^{k+1}F_{2j} &= \sum_{j=1}^k (F_{2j}) + F_{2(k+1)} \notag \\
                          &= (F_{2k+1} - 1) + F_{2(k+1)} \tag{by the inductive hypothesis} \\
                          &= F_{2k+1} + F_{2k+2} - 1 \notag \\
                          &= F_{2k+3} - 1 \tag{by the definition of $F_n$} \\
                          &= F_{2(k+1)+1} - 1 \notag
  \end{align}

  Once again, the substitution of $F_{2k+3}$ for $F_{2k+1} + F_{2k+2}$ is valid because the index $2k+3$ is always greater than 3.

  The result $F_{2(k+1)+1} - 1$ has the form $F_{2m+1} - 1$ for $m = k+1$. Thus, the result holds for the inductive case, and by the principle of mathematical induction the result holds for all $m \in \mathbb{N}$.

\end{proof}
