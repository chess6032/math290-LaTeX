\[
  F_n = \begin{cases}
    1                 & \text{if } n = 1 \\
    1                 & \text{if } n = 2 \\
    F_{n-1} + F_{n-2} & \text{if } n \ge 3
  \end{cases}
\]

\begin{proposition}
  \[
    \sum_{j=1}^{m} F_{2j-1} = F_{2m}
  \]
\end{proposition}

\begin{proof} The proof is by induction on $m$.

BASE CASE: Consider the case for $m=1$. Observe,

\[\begin{aligned}
  \sum_{j=1}^1 F_{2j-1} &= F_{2(1)-1} \\
                        &= F_{2-1} \\
                        &= F_1 \\
                        &= 1
\end{aligned}\]

and,

\[
  F_{2(1)} = F_2 = 1
\]

Thus, the result holds for the base case $m=1$.

INDUCTIVE CASE: Suppose the result holds for some $k$. That is,

\[
  \sum_{j=1}^{k} F_{2j-1} = F_{2k}
\]

\end{proof}

\begin{proposition}
  \[
    \sum_{j=1}^{m} F_{2j} = F_{2m+1} - 1
  \]
\end{proposition}

\begin{proof} The proof is by induction on $m$.

  An impressive proof.

\end{proof}
